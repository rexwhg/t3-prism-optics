function tick(now) {
  for (let i = 0; i < buckets.length; i += 1) buckets[i].clear();
  for (const item of activeItems) {
    const elapsed = now - item.startAt;
    if (elapsed >= TOTAL_MS) continue;
    const stage = Math.floor(elapsed / STEP_MS);
    const range = document.createRange();
    range.setStart(item.node, item.startOffset);
    range.setEnd(item.node, item.endOffset);
    buckets[stage].add(range);
  }
}
